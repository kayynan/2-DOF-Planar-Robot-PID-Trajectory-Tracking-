IK MATH — LAW OF COSINES
// Step 1: Solve θ₂
c₂ = (x² + y² − L₁² − L₂²) / (2·L₁·L₂)   s₂ = +√(1−c₂²)   θ₂ = atan2(s₂,c₂)

// Step 2: Solve θ₁
k₁ = L₁ + L₂·c₂   k₂ = L₂·s₂   θ₁ = atan2(y,x) − atan2(k₂,k₁)
